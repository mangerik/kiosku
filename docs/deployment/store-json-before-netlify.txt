CREATE OR REPLACE FUNCTION public.store_json(s stores, private_data boolean DEFAULT false) RETURNS jsonb LANGUAGE sql STABLE SET search_path TO 'public' AS $function$ select s.info || jsonb_build_object('id',s.id,'accountId',case when private_data then s.account_id::text else '' end,'name',s.name,'slug',s.slug,'status',s.status,'draft',case when private_data then s.draft else s.published end,'published',s.published,'createdAt',s.created_at,'versions',case when private_data then coalesce((select jsonb_agg(jsonb_build_object('at',l.created_at,'layout',l.layout_json) order by l.version desc) from store_layouts l where l.store_id=s.id),'[]'::jsonb) else '[]'::jsonb end) $function$
